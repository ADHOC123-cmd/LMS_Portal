1: // public/service-worker.js
2: // Service Worker with versioned cache and Network-First navigation.
3: const CACHE_NAME = "static-assets-v2";
4: 
5: self.addEventListener("install", () => {
6:   self.skipWaiting();
7: });
8: 
9: self.addEventListener("activate", (event) => {
10:   event.waitUntil(
11:     caches
12:       .keys()
13:       .then((keys) =>
14:         Promise.all(
15:           keys.map((key) => {
16:             if (key !== CACHE_NAME) {
17:               return caches.delete(key);
18:             }
19:           }),
20:         ),
21:       )
22:       .then(() => self.clients.claim()),
23:   );
24: });
25: 
26: self.addEventListener("fetch", (event) => {
27:   const url = new URL(event.request.url);
28: 
29:   // Ignore cross-origin and API requests
30:   if (url.origin !== location.origin || url.pathname.startsWith("/api")) {
31:     return;
32:   }
33: 
34:   // Network-First for HTML navigation / document requests
35:   // Ensures returning users immediately get the latest deployed site
36:   if (
37:     event.request.mode === "navigate" ||
38:     url.pathname.endsWith(".html") ||
39:     url.pathname === "/"
40:   ) {
41:     event.respondWith(
42:       fetch(event.request)
43:         .then((networkResponse) => {
44:           if (networkResponse.status === 200) {
45:             const clone = networkResponse.clone();
46:             caches.open(CACHE_NAME).then((cache) => cache.put(event.request, clone));
47:           }
48:           return networkResponse;
49:         })
50:         .catch(() => caches.match(event.request)),
51:     );
52:     return;
53:   }
54: 
55:   // Cache-First for static assets (hashed JS, CSS, images, etc.)
56:   if (event.request.method === "GET") {
57:     event.respondWith(
58:       caches.open(CACHE_NAME).then((cache) =>
59:         cache.match(event.request).then(
60:           (response) =>
61:             response ||
62:             fetch(event.request).then((networkResponse) => {
63:               if (networkResponse.status === 200) {
64:                 cache.put(event.request, networkResponse.clone());
65:               }
66:               return networkResponse;
67:             }),
68:         ),
69:       ),
70:     );
71:   }
72: });
--------------------------------------------------------------------------------

EXPLANATION:
  1. Old code used `static-assets` unversioned cache and blindly returned
     cached `index.html` via `response || fetch(event.request)`. That meant
     returning users NEVER received code updates from GitHub/Server unless
     they cleared their browser cache or used incognito.
  2. New code creates `static-assets-v2` and deletes older caches on activate.
  3. New code uses Network-First for navigation/HTML requests so users always
     receive the latest deployed HTML and JavaScript immediately upon release.
